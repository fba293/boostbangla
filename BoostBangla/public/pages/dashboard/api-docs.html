Code Examples

cURL

curl -X POST https://boostbangla.com/api/v2/order \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"service": 12345, "link": "https://instagram.com/p/xxx", "quantity": 1000}'

JavaScript (Fetch)

fetch('https://boostbangla.com/api/v2/order', {
  method: 'POST',
  headers: {
    'Authorization': 'Bearer YOUR_API_KEY',
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    service: 12345,
    link: 'https://instagram.com/p/xxx',
    quantity: 1000
  })
}).then(res => res.json())
  .then(data => console.log(data));

PHP (cURL)

$ch = curl_init('https://boostbangla.com/api/v2/order');
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'Authorization: Bearer YOUR_API_KEY',
    'Content-Type: application/json'
]);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode([
    'service' => 12345,
    'link' => 'https://instagram.com/p/xxx',
    'quantity' => 1000
]));
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
$response = curl_exec($ch);
$data = json_decode($response, true);

Python (Requests)

import requests

headers = {
    'Authorization': 'Bearer YOUR_API_KEY',
    'Content-Type': 'application/json'
}

data = {
    'service': 12345,
    'link': 'https://instagram.com/p/xxx',
    'quantity': 1000
}

response = requests.post(
    'https://boostbangla.com/api/v2/order',
    headers=headers,
    json=data
)
print(response.json())
